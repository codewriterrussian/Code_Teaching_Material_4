![Data Structure 四個核心問題](../images/lesson_34/CPP_Lesson_34_img02_four_questions_data_structure.png)


1. 每個資料結構先定義：
   ```text
   儲存什麼？
   ```

8. 節點 ownership 優先使用 `unique_ptr`。
9. 不用 undefined behavior 換取速度。
10. 所有合法完整程式使用嚴格 C++17 選項檢查。

## Section V. Invariant

![Invariant 是什麼？](../images/lesson_34/CPP_Lesson_34_img03_data_structure_invariant.png)


Invariant 是資料結構在每個 public operation 前後都必須成立的條件。

例如 circular queue：

```text
0 <= front < capacity
0 <= rear < capacity
0 <= count <= capacity
```

Linked list：

```text
head 為空代表 size == 0
每個 node 的 next 形成有限鏈
最後一個 node 的 next == nullptr
```

![Stack Pop / Top / Underflow](../images/lesson_34/CPP_Lesson_34_img08_stack_operations.png)


```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <stdexcept>
#include <utility>
#include <vector>
using namespace std;

template <typename T>
class Stack {
public:
    void push(
        const T& value
    ) {
        values.push_back(value);
    }

    void push(
        T&& value
    ) {
        values.push_back(
            std::move(value)
        );
    }

    void pop() {
        if (values.empty()) {
            throw underflow_error(
                "cannot pop an empty stack"
            );
        }

        values.pop_back();
    }

    T& top() {
        if (values.empty()) {
            throw underflow_error(
                "cannot read an empty stack"
            );
        }

        return values.back();
    }

    const T& top() const {
        if (values.empty()) {
            throw underflow_error(
                "cannot read an empty stack"
            );
        }

        return values.back();
    }

    bool empty() const noexcept {
        return values.empty();
    }

    size_t size() const noexcept {
        return values.size();
    }

private:
    vector<T> values;
};

int main() {
    Stack<int> stack;

    stack.push(10);
    stack.push(20);
    stack.push(30);

    cout << stack.top()
         << '\n';

    stack.pop();

    cout << stack.top()
         << " "
         << stack.size()
         << '\n';

    return 0;
}
```

## Section XIV. 完整 Circular Queue

![Circular Queue Enqueue 動畫式圖](../images/lesson_34/CPP_Lesson_34_img13_circular_queue_enqueue.png)

![Circular Queue Wraparound 實例](../images/lesson_34/CPP_Lesson_34_img14_circular_queue_physical_vs_logical.png)


```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <stdexcept>
#include <vector>
using namespace std;

template <typename T>
class CircularQueue {
public:
    explicit CircularQueue(
        size_t capacity
    )
        : values(capacity),
          capacityValue(capacity) {
        if (capacity == 0) {
            throw invalid_argument(
                "capacity must be positive"
            );
        }
    }

    void enqueue(
        const T& value
    ) {
        if (full()) {
            throw overflow_error(
                "queue is full"
            );
        }

        values[rearIndex] =
            value;

        rearIndex =
            (
                rearIndex +
                1
            ) %
            capacityValue;

        ++countValue;
    }

    void dequeue() {
        if (empty()) {
            throw underflow_error(
                "queue is empty"
            );
        }

        frontIndex =
            (
                frontIndex +
                1
            ) %
            capacityValue;

        --countValue;
    }

    const T& front() const {
        if (empty()) {
            throw underflow_error(
                "queue is empty"
            );
        }

        return values[frontIndex];
    }

    bool empty() const noexcept {
        return countValue == 0;
    }

    bool full() const noexcept {
        return
            countValue ==
            capacityValue;
    }

    size_t size() const noexcept {
        return countValue;
    }

private:
    vector<T> values;
    size_t capacityValue;
    size_t frontIndex = 0;
    size_t rearIndex = 0;
    size_t countValue = 0;
};

int main() {
    CircularQueue<int> queue(3);

    queue.enqueue(10);
    queue.enqueue(20);
    queue.enqueue(30);

    cout << queue.front()
         << '\n';

    queue.dequeue();
    queue.enqueue(40);

    while (!queue.empty()) {
        cout << queue.front()
             << " ";

        queue.dequeue();
    }

    cout << '\n';

    return 0;
}
```

![Reverse Linked List](../images/lesson_34/CPP_Lesson_34_img23_reverse_singly_linked_list.png)

![Why Save `next` First?](../images/lesson_34/CPP_Lesson_34_img24_reverse_list_pointer_order.png)


```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <memory>
#include <utility>
using namespace std;

class SinglyLinkedList {
private:
    struct Node {
        explicit Node(int initialValue)
            : value(initialValue) {
        }

        int value;
        unique_ptr<Node> next;
    };

public:
    void pushFront(int value) {
        auto node =
            make_unique<Node>(
                value
            );

        node->next =
            std::move(head);

        head =
            std::move(node);

        ++sizeValue;
    }

    void pushBack(int value) {
        auto node =
            make_unique<Node>(
                value
            );

        if (!head) {
            head =
                std::move(node);

            ++sizeValue;
            return;
        }

        Node* current =
            head.get();

        while (current->next) {
            current =
                current->next.get();
        }

        current->next =
            std::move(node);

        ++sizeValue;
    }

    bool contains(int value) const {
        const Node* current =
            head.get();

        while (current != nullptr) {
            if (
                current->value ==
                value
            ) {
                return true;
            }

            current =
                current->next.get();
        }

        return false;
    }

    bool eraseFirst(int value) {
        unique_ptr<Node>* link =
            &head;

        while (*link) {
            if (
                (*link)->value ==
                value
            ) {
                *link =
                    std::move(
                        (*link)->next
                    );

                --sizeValue;
                return true;
            }

            link =
                &(
                    (*link)->next
                );
        }

        return false;
    }

    void reverse() noexcept {
        unique_ptr<Node> previous;

        while (head) {
            unique_ptr<Node> next =
                std::move(
                    head->next
                );

            head->next =
                std::move(
                    previous
                );

            previous =
                std::move(
                    head
                );

            head =
                std::move(
                    next
                );
        }

        head =
            std::move(
                previous
            );
    }

    size_t size() const noexcept {
        return sizeValue;
    }

    void print() const {
        const Node* current =
            head.get();

        while (current != nullptr) {
            cout << current->value
                 << " ";

            current =
                current->next.get();
        }

        cout << '\n';
    }

private:
    unique_ptr<Node> head;
    size_t sizeValue = 0;
};

int main() {
    SinglyLinkedList list;

    list.pushBack(10);
    list.pushBack(20);
    list.pushFront(5);

    list.print();

    list.eraseFirst(10);
    list.reverse();
    list.print();

    cout << boolalpha
         << list.contains(20)
         << " "
         << list.size()
         << '\n';

    return 0;
}
```

# Part G：Doubly Linked List

![Doubly Linked List](../images/lesson_34/CPP_Lesson_34_img26_doubly_linked_list.png)


## Section XX. 結構

每個 node 有：

```text
previous ← node → next
```

優點：

- 可雙向走訪。
- 已知 node 時較容易刪除。
- 可 O(1) 操作兩端，前提是保存 head 與 tail。

代價：

- 每個 node 多一個 pointer。
- Pointer invariant 更複雜。

## Section XXI. Invariant

![Doubly List Pointer Invariant](../images/lesson_34/CPP_Lesson_34_img28_doubly_linked_invariant.png)


空串列：

```text
head == nullptr
tail == nullptr
size == 0
```

非空串列：

```text
head->previous == nullptr
tail->next == nullptr
```

每對相鄰節點：

```text
node->next->previous == node
```

```cpp
// VALIDATE
#include <algorithm>
#include <iostream>
#include <memory>
#include <utility>
using namespace std;

class BinarySearchTree {
private:
    struct Node {
        explicit Node(int initialKey)
            : key(initialKey) {
        }

        int key;
        unique_ptr<Node> left;
        unique_ptr<Node> right;
    };

public:
    bool insert(int key) {
        return insertNode(
            root,
            key
        );
    }

    bool contains(int key) const {
        const Node* current =
            root.get();

        while (current != nullptr) {
            if (key == current->key) {
                return true;
            }

            if (key < current->key) {
                current =
                    current->left.get();
            } else {
                current =
                    current->right.get();
            }
        }

        return false;
    }

    bool erase(int key) {
        return eraseNode(
            root,
            key
        );
    }

    int height() const noexcept {
        return heightOf(
            root.get()
        );
    }

    void printInorder() const {
        printInorderNode(
            root.get()
        );

        cout << '\n';
    }

private:
    static bool insertNode(
        unique_ptr<Node>& node,
        int key
    ) {
        if (!node) {
            node =
                make_unique<Node>(
                    key
                );

            return true;
        }

        if (key < node->key) {
            return insertNode(
                node->left,
                key
            );
        }

        if (key > node->key) {
            return insertNode(
                node->right,
                key
            );
        }

        return false;
    }

    static bool eraseNode(
        unique_ptr<Node>& node,
        int key
    ) {
        if (!node) {
            return false;
        }

        if (key < node->key) {
            return eraseNode(
                node->left,
                key
            );
        }

        if (key > node->key) {
            return eraseNode(
                node->right,
                key
            );
        }

        if (!node->left) {
            node =
                std::move(
                    node->right
                );

            return true;
        }

        if (!node->right) {
            node =
                std::move(
                    node->left
                );

            return true;
        }

        Node* successor =
            node->right.get();

        while (successor->left) {
            successor =
                successor->left.get();
        }

        node->key =
            successor->key;

        return eraseNode(
            node->right,
            successor->key
        );
    }

    static int heightOf(
        const Node* node
    ) noexcept {
        if (node == nullptr) {
            return 0;
        }

        return
            1 +
            max(
                heightOf(
                    node->left.get()
                ),
                heightOf(
                    node->right.get()
                )
            );
    }

    static void printInorderNode(
        const Node* node
    ) {
        if (node == nullptr) {
            return;
        }

        printInorderNode(
            node->left.get()
        );

        cout << node->key
             << " ";

        printInorderNode(
            node->right.get()
        );
    }

    unique_ptr<Node> root;
};

int main() {
    BinarySearchTree tree;

    for (
        int key :
        {
            8,
            3,
            10,
            1,
            6,
            14,
            4,
            7,
            13
        }
    ) {
        tree.insert(key);
    }

    tree.printInorder();

    tree.erase(3);
    tree.printInorder();

    cout << boolalpha
         << tree.contains(7)
         << " "
         << tree.height()
         << '\n';

    return 0;
}
```

## Section XXXI. 完整 Min Heap

```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <stdexcept>
#include <utility>
#include <vector>
using namespace std;

class MinHeap {
public:
    void push(int value) {
        values.push_back(value);

        siftUp(
            values.size() -
            1
        );
    }

    int top() const {
        if (values.empty()) {
            throw underflow_error(
                "heap is empty"
            );
        }

        return values.front();
    }

    void pop() {
        if (values.empty()) {
            throw underflow_error(
                "heap is empty"
            );
        }

        values.front() =
            values.back();

        values.pop_back();

        if (!values.empty()) {
            siftDown(0);
        }
    }

    bool empty() const noexcept {
        return values.empty();
    }

    size_t size() const noexcept {
        return values.size();
    }

private:
    void siftUp(size_t index) {
        while (index > 0) {
            const size_t parent =
                (
                    index -
                    1
                ) /
                2;

            if (
                values[parent] <=
                values[index]
            ) {
                return;
            }

            swap(
                values[parent],
                values[index]
            );

            index =
                parent;
        }
    }

    void siftDown(size_t index) {
        while (true) {
            const size_t left =
                index *
                    2 +
                1;

            const size_t right =
                index *
                    2 +
                2;

            size_t smallest =
                index;

            if (
                left <
                    values.size() &&
                values[left] <
                    values[smallest]
            ) {
                smallest =
                    left;
            }

            if (
                right <
                    values.size() &&
                values[right] <
                    values[smallest]
            ) {
                smallest =
                    right;
            }

            if (
                smallest ==
                index
            ) {
                return;
            }

            swap(
                values[index],
                values[smallest]
            );

            index =
                smallest;
        }
    }

    vector<int> values;
};

int main() {
    MinHeap heap;

    for (
        int value :
        {
            7,
            2,
            9,
            1,
            5
        }
    ) {
        heap.push(value);
    }

    while (!heap.empty()) {
        cout << heap.top()
             << " ";

        heap.pop();
    }

    cout << '\n';

    return 0;
}
```

## Section XXXVI. 完整 Chained Hash Table

![Rehash](../images/lesson_34/CPP_Lesson_34_img50_hash_rehash.png)


```cpp
// VALIDATE
#include <cstddef>
#include <functional>
#include <iostream>
#include <list>
#include <optional>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>
using namespace std;

template <
    typename Key,
    typename Value
>
class ChainedHashTable {
private:
    using Entry =
        pair<Key, Value>;

public:
    explicit ChainedHashTable(
        size_t bucketCount = 8
    )
        : buckets(bucketCount) {
        if (bucketCount == 0) {
            throw invalid_argument(
                "bucket count must be positive"
            );
        }
    }

    void insertOrAssign(
        const Key& key,
        const Value& value
    ) {
        maybeRehash();

        auto& bucket =
            buckets[
                indexFor(key)
            ];

        for (auto& entry : bucket) {
            if (entry.first == key) {
                entry.second =
                    value;

                return;
            }
        }

        bucket.emplace_back(
            key,
            value
        );

        ++sizeValue;
    }

    optional<Value> find(
        const Key& key
    ) const {
        const auto& bucket =
            buckets[
                indexFor(key)
            ];

        for (
            const auto& entry :
            bucket
        ) {
            if (entry.first == key) {
                return entry.second;
            }
        }

        return nullopt;
    }

    bool erase(
        const Key& key
    ) {
        auto& bucket =
            buckets[
                indexFor(key)
            ];

        for (
            auto iterator =
                bucket.begin();
            iterator !=
                bucket.end();
            ++iterator
        ) {
            if (
                iterator->first ==
                key
            ) {
                bucket.erase(
                    iterator
                );

                --sizeValue;
                return true;
            }
        }

        return false;
    }

    size_t size() const noexcept {
        return sizeValue;
    }

    double loadFactor() const {
        return
            static_cast<double>(
                sizeValue
            ) /
            static_cast<double>(
                buckets.size()
            );
    }

private:
    size_t indexFor(
        const Key& key
    ) const {
        return
            hash<Key>{}(key) %
            buckets.size();
    }

    void maybeRehash() {
        if (
            loadFactor() <
            0.75
        ) {
            return;
        }

        vector<
            list<Entry>
        > newBuckets(
            buckets.size() *
            2
        );

        for (
            const auto& bucket :
            buckets
        ) {
            for (
                const auto& entry :
                bucket
            ) {
                const size_t index =
                    hash<Key>{}(
                        entry.first
                    ) %
                    newBuckets.size();

                newBuckets[index]
                    .push_back(entry);
            }
        }

        buckets =
            std::move(
                newBuckets
            );
    }

    vector<
        list<Entry>
    > buckets;

    size_t sizeValue = 0;
};

int main() {
    ChainedHashTable<
        string,
        int
    > table(2);

    table.insertOrAssign(
        "Amy",
        95
    );

    table.insertOrAssign(
        "Brian",
        88
    );

    table.insertOrAssign(
        "Amy",
        100
    );

    const optional<int> score =
        table.find("Amy");

    if (score) {
        cout << *score
             << '\n';
    }

    table.erase("Brian");

    cout << table.size()
         << " "
         << table.loadFactor()
         << '\n';

    return 0;
}
```

```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <queue>
#include <stdexcept>
#include <vector>
using namespace std;

class Graph {
public:
    explicit Graph(
        size_t vertexCount
    )
        : adjacency(
              vertexCount
          ) {
    }

    void addUndirectedEdge(
        size_t first,
        size_t second
    ) {
        validateVertex(first);
        validateVertex(second);

        adjacency[first]
            .push_back(second);

        adjacency[second]
            .push_back(first);
    }

    vector<int> distancesFrom(
        size_t start
    ) const {
        validateVertex(start);

        vector<int> distance(
            adjacency.size(),
            -1
        );

        queue<size_t> pending;

        distance[start] = 0;
        pending.push(start);

        while (!pending.empty()) {
            const size_t current =
                pending.front();

            pending.pop();

            for (
                size_t neighbor :
                adjacency[current]
            ) {
                if (
                    distance[neighbor] !=
                    -1
                ) {
                    continue;
                }

                distance[neighbor] =
                    distance[current] +
                    1;

                pending.push(
                    neighbor
                );
            }
        }

        return distance;
    }

private:
    void validateVertex(
        size_t vertex
    ) const {
        if (
            vertex >=
            adjacency.size()
        ) {
            throw out_of_range(
                "invalid vertex"
            );
        }
    }

    vector<
        vector<size_t>
    > adjacency;
};

int main() {
    Graph graph(6);

    graph.addUndirectedEdge(0, 1);
    graph.addUndirectedEdge(0, 2);
    graph.addUndirectedEdge(1, 3);
    graph.addUndirectedEdge(2, 4);

    const vector<int> distance =
        graph.distancesFrom(0);

    for (int value : distance) {
        cout << value
             << " ";
    }

    cout << '\n';

    return 0;
}
```

```cpp
// VALIDATE
#include <cstddef>
#include <iostream>
#include <stack>
#include <stdexcept>
#include <vector>
using namespace std;

vector<size_t> depthFirstOrder(
    const vector<
        vector<size_t>
    >& adjacency,
    size_t start
) {
    if (
        start >=
        adjacency.size()
    ) {
        throw out_of_range(
            "invalid start vertex"
        );
    }

    vector<bool> visited(
        adjacency.size(),
        false
    );

    vector<size_t> order;
    stack<size_t> pending;

    pending.push(start);

    while (!pending.empty()) {
        const size_t current =
            pending.top();

        pending.pop();

        if (visited[current]) {
            continue;
        }

        visited[current] =
            true;

        order.push_back(
            current
        );

        const auto& neighbors =
            adjacency[current];

        for (
            auto iterator =
                neighbors.rbegin();
            iterator !=
                neighbors.rend();
            ++iterator
        ) {
            if (!visited[*iterator]) {
                pending.push(
                    *iterator
                );
            }
        }
    }

    return order;
}

int main() {
    const vector<
        vector<size_t>
    > graph{
        {1, 2},
        {3},
        {4},
        {},
        {}
    };

    const vector<size_t> order =
        depthFirstOrder(
            graph,
            0
        );

    for (size_t vertex : order) {
        cout << vertex
             << " ";
    }

    cout << '\n';

    return 0;
}
```

## Section XLVIII. 完整 Dijkstra

![Dijkstra Priority Queue](../images/lesson_34/CPP_Lesson_34_img68_dijkstra_priority_queue.png)

![Stale Entry Check](../images/lesson_34/CPP_Lesson_34_img69_dijkstra_stale_entry.png)


```cpp
// VALIDATE
#include <cstddef>
#include <functional>
#include <iostream>
#include <limits>
#include <queue>
#include <stdexcept>
#include <utility>
#include <vector>
using namespace std;

struct WeightedEdge {
    size_t to;
    int weight;
};

vector<long long> dijkstra(
    const vector<
        vector<WeightedEdge>
    >& graph,
    size_t start
) {
    if (
        start >=
        graph.size()
    ) {
        throw out_of_range(
            "invalid start vertex"
        );
    }

    for (
        const auto& edges :
        graph
    ) {
        for (
            const WeightedEdge& edge :
            edges
        ) {
            if (edge.weight < 0) {
                throw invalid_argument(
                    "negative edge weight"
                );
            }

            if (
                edge.to >=
                graph.size()
            ) {
                throw out_of_range(
                    "invalid edge endpoint"
                );
            }
        }
    }

    const long long infinity =
        numeric_limits<
            long long
        >::max();

    vector<long long> distance(
        graph.size(),
        infinity
    );

    using State =
        pair<
            long long,
            size_t
        >;

    priority_queue<
        State,
        vector<State>,
        greater<State>
    > pending;

    distance[start] = 0;
    pending.push(
        {
            0,
            start
        }
    );

    while (!pending.empty()) {
        const auto [
            currentDistance,
            current
        ] =
            pending.top();

        pending.pop();

        if (
            currentDistance !=
            distance[current]
        ) {
            continue;
        }

        for (
            const WeightedEdge& edge :
            graph[current]
        ) {
            const long long candidate =
                currentDistance +
                static_cast<long long>(
                    edge.weight
                );

            if (
                candidate <
                distance[edge.to]
            ) {
                distance[edge.to] =
                    candidate;

                pending.push(
                    {
                        candidate,
                        edge.to
                    }
                );
            }
        }
    }

    return distance;
}

int main() {
    const vector<
        vector<WeightedEdge>
    > graph{
        {
            {1, 4},
            {2, 1}
        },
        {
            {3, 1}
        },
        {
            {1, 2},
            {3, 5}
        },
        {}
    };

    const vector<long long> distance =
        dijkstra(
            graph,
            0
        );

    for (long long value : distance) {
        cout << value
             << " ";
    }

    cout << '\n';

    return 0;
}
```

## Section LX. Ownership 設計


## Section LXII. Push Back 步驟

空串列：

```text
head owns new node
tail observes head
```

非空串列：

```text
new.previous = tail
tail.next owns new
tail = tail.next
```

---

## Section LXIII. Pop Back 難點

因為 ownership 由前一節點的 `next` 持有，刪除 tail 時要：

1. 找到 `tail->previous`
2. 將 previous 的 `next.reset()`
3. 更新 tail
4. 若刪除最後一個節點，head 與 tail 都清空

# Part AE：常見錯誤提醒

## Section LXXXII. 常見錯誤

1. 沒有先定義 invariant。
2. Stack 空時仍呼叫 top。
3. Queue 空時仍 dequeue。
4. Circular queue 無法區分 full 與 empty。
5. Modulo 使用錯誤 capacity。
6. Rear 更新順序錯誤。
7. Linked list 插入時先覆寫 next，造成後半段遺失。
8. 刪除 node 後仍使用其 pointer。
9. 忘記更新 head。
10. 保存 tail 時忘記更新 tail。
11. Doubly linked list 只更新 next，忘記 previous。
12. 空串列時 head／tail 狀態不一致。
13. BST duplicate policy 不清楚。
14. BST 刪除 two-child node 時 successor 處理錯誤。
15. BST 高度定義不一致。
16. 把 heap 當成完全排序陣列。
17. Heap parent／child index off-by-one。
18. Pop heap 後沒有 sift down。
19. Hash table 忽略 collision。
20. Rehash 後忘記重新計算 bucket index。
21. Linear probing 刪除時直接設 empty。
22. Load factor 使用整數除法。
23. Graph edge endpoint 未驗證。
24. Undirected edge 只加入一個方向。
25. BFS 沒在 enqueue 時標記 visited，造成重複加入。
26. DFS recursion 太深造成 stack overflow。
27. Dijkstra 用於負權重。
28. Dijkstra 沒跳過 stale entry。
29. 距離加法 overflow。
30. 只測一般情況，未測空結構與邊界。

# Part AF：Mermaid 流程圖

## Section LXXXIII. 資料結構流程圖

### 1. 選擇資料結構

```mermaid
flowchart TD
    A[分析主要操作] --> B{需要 LIFO 嗎}
    B -- 是 --> C[Stack]
    B -- 否 --> D{需要 FIFO 嗎}
    D -- 是 --> E[Queue]
    D -- 否 --> F{需要 key lookup 嗎}
    F -- 是 --> G{需要排序嗎}
    G -- 是 --> H[Tree Map]
    G -- 否 --> I[Hash Table]
    F -- 否 --> J{資料是節點與邊嗎}
    J -- 是 --> K[Graph]
    J -- 否 --> L[Vector or Deque]
```

### 2. Circular Queue Enqueue

```mermaid
flowchart TD
    A[Enqueue value] --> B{Queue full}
    B -- 是 --> C[Reject or throw]
    B -- 否 --> D[Write at rear]
    D --> E[rear equals rear plus 1 modulo capacity]
    E --> F[Increase count]
```

### 3. Singly Linked List Insert Front

```mermaid
flowchart TD
    A[Create new node] --> B[new next owns old head]
    B --> C[head owns new node]
    C --> D[Increase size]
```

### 4. BST Insert

```mermaid
flowchart TD
    A[Start at root] --> B{Current is null}
    B -- 是 --> C[Create new leaf]
    B -- 否 --> D{Key smaller}
    D -- 是 --> E[Move to left child]
    D -- 否 --> F{Key larger}
    F -- 是 --> G[Move to right child]
    F -- 否 --> H[Duplicate rejected]
    E --> B
    G --> B
```

### 5. Heap Push

```mermaid
flowchart TD
    A[Append value at end] --> B[Compare with parent]
    B --> C{Heap order valid}
    C -- 是 --> D[Finish]
    C -- 否 --> E[Swap with parent]
    E --> B
```

### 6. Hash Table Lookup

```mermaid
flowchart TD
    A[Hash key] --> B[Choose bucket]
    B --> C[Scan bucket entries]
    C --> D{Key found}
    D -- 是 --> E[Return value]
    D -- 否 --> F{More entries}
    F -- 是 --> C
    F -- 否 --> G[Return missing]
```

### 7. BFS

```mermaid
flowchart TD
    A[Mark start visited and enqueue] --> B{Queue empty}
    B -- 是 --> C[Finish]
    B -- 否 --> D[Dequeue current]
    D --> E[Inspect each neighbor]
    E --> F{Neighbor unvisited}
    F -- 是 --> G[Mark visited and enqueue]
    F -- 否 --> H[Skip]
    G --> E
    H --> E
    E --> B
```

### 8. Dijkstra

```mermaid
flowchart TD
    A[distance start equals zero] --> B[Push start into min priority queue]
    B --> C{Queue empty}
    C -- 是 --> D[Finish distances]
    C -- 否 --> E[Pop smallest distance state]
    E --> F{State stale}
    F -- 是 --> C
    F -- 否 --> G[Relax outgoing edges]
    G --> H{Candidate shorter}
    H -- 是 --> I[Update distance and push state]
    H -- 否 --> J[Keep current distance]
    I --> G
    J --> G
    G --> C
```
